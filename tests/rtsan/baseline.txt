# RealTimeSanitizer baseline: real-time violations inside third-party code (DECISIONS W12).
#
# Format: <unsafe call RTSan intercepted> | <innermost Tracktion/JUCE frame that owns it>
# scripts/rtsan_check.py fails on any report not listed here, and on any report owned by Carthage's
# code (carthage::), which can never be listed. Recorded 2026-11-17 against JUCE 9.0.5 and Tracktion
# Engine develop c6293127 (DECISIONS W7) with Neural Amp Modeler 0.7.9 as VST3 and AU, and with no
# plugin. Re-record when a pin changes, and delete entries upstream has fixed.

# --- Tracktion Engine: the device callback, every block -------------------------------------------
# contextLock (std::shared_mutex) around the active playback contexts.
pthread_mutex_lock   | tracktion::engine::DeviceManager::audioDeviceIOCallbackInternal
pthread_mutex_unlock | tracktion::engine::DeviceManager::audioDeviceIOCallbackInternal
# midiInputsMutex (std::shared_mutex), and MIDI input threads woken with a condition variable.
pthread_mutex_lock   | tracktion::engine::DeviceManager::broadcastStreamTimeToMidiDevices
pthread_mutex_unlock | tracktion::engine::DeviceManager::broadcastStreamTimeToMidiDevices
pthread_cond_signal  | tracktion::engine::DeviceManager::broadcastStreamTimeToMidiDevices
pthread_mutex_lock   | tracktion::engine::MidiInputDevice::masterTimeUpdate

# --- Tracktion Engine: the live input path, every block -------------------------------------------
# consumerLock (CriticalSection) while handing the input to the graph.
pthread_mutex_lock   | tracktion::engine::WaveInputDevice::consumeNextAudioBlock
pthread_mutex_unlock | tracktion::engine::WaveInputDevice::consumeNextAudioBlock
pthread_mutex_lock   | tracktion::engine::WaveInputDeviceInstance::acceptInputBuffer
pthread_mutex_unlock | tracktion::engine::WaveInputDeviceInstance::acceptInputBuffer
# The input copy buffer is resized from the audio thread when the channel count changes.
malloc               | tracktion::engine::WaveInputDeviceInstance::copyIncomingDataIntoBuffer
free                 | tracktion::engine::WaveInputDeviceInstance::copyIncomingDataIntoBuffer
malloc               | tracktion::engine::WaveInputDevice::consumeNextAudioBlock
free                 | tracktion::engine::WaveInputDevice::consumeNextAudioBlock
# The retrospective-record buffer (always created for physical inputs) resizes itself and asks the
# device for its rate and buffer size every block; JUCE's CoreAudio device answers by copying its
# DeviceSetup, which holds BigIntegers: a malloc and a free per call.
malloc               | tracktion::engine::AudioFifo::setSize
free                 | tracktion::engine::AudioFifo::setSize
malloc               | tracktion::engine::RetrospectiveRecordBuffer::syncToEdit
malloc               | juce::DeviceConfigurator::getCurrentSetup
free                 | juce::CoreAudioClasses::CoreAudioIODevice::getCurrentSampleRate
free                 | juce::CoreAudioClasses::CoreAudioIODevice::getCurrentBufferSizeSamples
